# ====================================================
# PROGRAMA: Bucle de Cuenta Regresiva (Loop)
# ====================================================
# Este código demuestra cómo el procesador realiza 
# tareas repetitivas saltando hacia atrás en la memoria.
# ====================================================

# 1. Preparación (PC: 0x00, 0x04, 0x08)
li t0, 5            # t0 = 5 (Nuestro contador, dará 5 vueltas)
li t1, 0            # t1 = 0 (El límite con el que compararemos)
li t2, 100          # t2 = 100 (Iremos sumando de 10 en 10 aquí)

# 2. Cuerpo del Bucle (PC: 0x0C, 0x10)
# El procesador repetirá estas dos instrucciones:
addi t0, t0, -1     # Restamos 1 al contador (t0)
addi t2, t2, 10     # Le sumamos 10 a nuestra variable (t2)

# 3. Condición de Salto (PC: 0x14)
# bne = "Branch if Not Equal" (Salta si NO son iguales)
# Si t0 no es igual a t1(0), salta -8 bytes hacia atrás.
# Retroceder 8 bytes significa volver exactamente al primer 'addi'.
bne t0, t1, -8      

# FIN DEL PROGRAMA
